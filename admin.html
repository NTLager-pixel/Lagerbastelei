// ========================================== // KONSTANTEN // ========================================== const SHEET_ARTIKEL = 'Artikel'; const SHEET_LAGER = 'Lagerplaetze'; const SHEET_BEWEGUNGEN = 'Bewegungen'; const SHEET_CONFIG = 'Konfiguration'; // ========================================== // API-SCHNITTSTELLE (POST) // ========================================== function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return responseJSON({ success: false, error: 'Keine Daten empfangen.' }); } const data = JSON.parse(e.postData.contents); const sheetConfig = SpreadsheetApp.getActive().getSheetByName(SHEET_CONFIG); const configData = sheetConfig ? sheetConfig.getDataRange().getValues() : []; let viewPasswortImSheet = ''; let passwortImSheet = ''; let adminPasswortImSheet = ''; for (let i = 0; i < configData.length; i++) { const key = String(configData[i][0]).trim().toUpperCase(); const val = String(configData[i][1]).trim(); if (key === 'VIEW_PASSWORT') viewPasswortImSheet = val; if (key === 'APP_PASSWORT') passwortImSheet = val; if (key === 'ADMIN_PASSWORT') adminPasswortImSheet = val; } const action = data.action; // 1. Admin-Login Prüfaktion if (action === 'checkAdmin') { const inputPw = data.adminPw ? String(data.adminPw).trim() : ''; const isValid = (inputPw !== '' && inputPw === adminPasswortImSheet); return responseJSON({ success: isValid }); } // 2. User/View-Token Prüfaktion if (action === 'checkToken') { const inputSecret = data.secret ? String(data.secret).trim() : ''; const isValid = (inputSecret !== '' && (inputSecret === passwortImSheet || inputSecret === viewPasswortImSheet || inputSecret === adminPasswortImSheet)); return responseJSON({ success: isValid }); } // Rechteprüfung für geschützte Aktionen const reqSecret = data.secret ? String(data.secret).trim() : ''; const reqAdminPw = data.adminPw ? String(data.adminPw).trim() : ''; const isView = (reqSecret !== '' && reqSecret === viewPasswortImSheet); const isUser = (reqSecret !== '' && reqSecret === passwortImSheet); const isAdmin = (reqAdminPw !== '' && reqAdminPw === adminPasswortImSheet) || (reqSecret !== '' && reqSecret === adminPasswortImSheet); if (!isView && !isUser && !isAdmin) { return responseJSON({ success: false, error: 'Zugriff verweigert: Ungültige Authentifizierung.' }); } if (action === 'createArtikel') { if (!isAdmin) return responseJSON({ success: false, error: 'Nur für Admins gestattet.' }); const artikelId = "ART-" + Date.now().toString().slice(-6); const res = erstelleNeuenArtikel(artikelId, data.artikelname, data.code, data.lagerplatzId, data.bestand); return responseJSON(res); } if (action === 'updateArtikel') { if (!isAdmin) return responseJSON({ success: false, error: 'Nur für Admins gestattet.' }); const res = updateArtikelData(data.artikelId, data.artikelname, data.code, data.lagerplatzId, data.bestand); return responseJSON(res); } if (action === 'suche') { // Sowohl View, User als auch Admin dürfen suchen/anzeigen const ergebnisse = sucheArtikelMitLagerplatz(data.query); return responseJSON({ success: true, treffer: ergebnisse }); } if (action === 'getArtikel') { const code = String(data.code || '').trim(); const artikelListe = getArtikelListe(); const lagerplaetze = getLagerplaetze(); const lagerMatch = lagerplaetze.find(l => String(l.lagerplatzId).toLowerCase() === code.toLowerCase() || String(l.bezeichnung).toLowerCase() === code.toLowerCase() ); if (lagerMatch) { const treffer = sucheArtikelMitLagerplatz(lagerMatch.lagerplatzId); return responseJSON({ success: true, isLagerplatz: true, treffer: treffer, lagerplatzName: lagerMatch.bezeichnung }); } let artikel = artikelListe.find(a => String(a.barcode) === code || String(a.qrcode) === code || String(a.artikelId) === code ) || null; if (artikel) { const lager = lagerplaetze.find(l => String(l.lagerplatzId) === String(artikel.lagerplatzId)); artikel.lagerplatzBeschreibung = lager ? lager.beschreibung : ''; artikel.lagerplatzName = lager ? lager.bezeichnung : artikel.lagerplatzId; return responseJSON({ success: true, isLagerplatz: false, artikel: artikel }); } return responseJSON({ success: false, error: "Code '" + code + "' weder als Artikel noch als Lagerplatz gefunden." }); } if (action === 'buchen') { // Reine View-User dürfen nicht buchen if (isView) { return responseJSON({ success: false, error: 'Nur Lesezugriff: Buchungen sind nicht gestattet.' }); } if (!isAdmin && data.typ !== 'ENTNAHME') { return responseJSON({ success: false, error: 'Nur Entnahmen sind für normale User gestattet.' }); } const benutzerName = isAdmin ? 'Admin' : 'Lager-User'; if (data.typ === 'ENTNAHME') { const result = entnahmeBuchen(data.artikelId, data.menge, benutzerName, 'App'); return responseJSON({ success: true, result: result }); } else if (data.typ === 'EINLAGERUNG') { const result = einlagerungBuchen(data.artikelId, data.menge, benutzerName, 'App'); return responseJSON({ success: true, result: result }); } } return responseJSON({ success: false, error: 'Unbekannte Aktion' }); } catch (err) { return responseJSON({ success: false, error: err.message }); } } function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ========================================== // HILFSFUNKTIONEN (INTERN) // ========================================== function getSheetData_(sheetName) { const sheet = SpreadsheetApp.getActive().getSheetByName(sheetName); if (!sheet) return []; const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; data.shift(); return data; } function getArtikelListe() { const data = getSheetData_(SHEET_ARTIKEL); return data .filter(row => row && row[0] !== undefined && String(row[0]).trim() !== '') .map(row => ({ artikelId: String(row[0]), artikelname: String(row[1] || ''), barcode: String(row[2] || ''), qrcode: String(row[3] || ''), lagerplatzId: String(row[4] || ''), bestand: Number(row[5]) || 0, mindestbestand: Number(row[6]) || 0, aktiv: String(row[7] || 'JA') })); } function getLagerplaetze() { const data = getSheetData_(SHEET_LAGER); return data .filter(row => row && row[0] !== undefined && String(row[0]).trim() !== '') .map(row => ({ lagerplatzId: String(row[0]), bezeichnung: String(row[1] || ''), beschreibung: String(row[2] || ''), aktiv: String(row[3] || 'JA') })); } function sucheArtikelMitLagerplatz(suchtext) { const begriff = String(suchtext || '').trim(); if (!begriff) return []; const alleArtikel = getArtikelListe(); const lagerplaetze = getLagerplaetze(); if (begriff === '*') { return alleArtikel.map(a => { const lager = lagerplaetze.find(l => String(l.lagerplatzId) === String(a.lagerplatzId)); return { artikelId: a.artikelId, artikelname: a.artikelname, bestand: a.bestand, lagerplatzId: a.lagerplatzId, lagerplatzName: lager ? lager.bezeichnung : a.lagerplatzId, lagerplatzBeschreibung: lager ? lager.beschreibung : '' }; }); } // Suchbegriffe in einzelne Wörter zerlegen const searchTokens = begriff.toLowerCase() .replace(/[^a-z0-9äöüß\s]/g, "") .split(/\s+/) .filter(t => t.length > 0); return alleArtikel .filter(a => { const lager = lagerplaetze.find(l => String(l.lagerplatzId) === String(a.lagerplatzId)); const lagerName = lager ? lager.bezeichnung : ''; const lagerDesc = lager ? lager.beschreibung : ''; // Durchsuchbarer Text pro Artikel const fullText = `${a.artikelId} ${a.artikelname} ${a.barcode} ${a.qrcode} ${a.lagerplatzId} ${lagerName} ${lagerDesc}`.toLowerCase(); const targetTokens = fullText.replace(/[^a-z0-9äöüß\s]/g, "").split(/\s+/).filter(t => t.length > 0); // Prüfen, ob jedes gesuchte Wort im Artikel vorhanden ist return searchTokens.every(sToken => { return targetTokens.some(tToken => { if (tToken.includes(sToken) || sToken.includes(tToken)) return true; if (sToken.length > 3 && tToken.length > 3) { return levenshteinDistance(sToken, tToken) <= 2; } return false; }); }); }) .map(a => { const lager = lagerplaetze.find(l => String(l.lagerplatzId) === String(a.lagerplatzId)); return { artikelId: a.artikelId, artikelname: a.artikelname, bestand: a.bestand, lagerplatzId: a.lagerplatzId, lagerplatzName: lager ? lager.bezeichnung : a.lagerplatzId, lagerplatzBeschreibung: lager ? lager.beschreibung : '' }; }); } function levenshteinDistance(a, b) { if (a.length === 0) return b.length; if (b.length === 0) return a.length; const matrix = []; for (let i = 0; i <= b.length; i++) matrix[i] = [i]; for (let j = 0; j <= a.length; j++) matrix[0][j] = j; for (let i = 1; i <= b.length; i++) { for (let j = 1; j <= a.length; j++) { if (b.charAt(i - 1) === a.charAt(j - 1)) { matrix[i][j] = matrix[i - 1][j - 1]; } else { matrix[i][j] = Math.min( matrix[i - 1][j - 1] + 1, matrix[i][j - 1] + 1, matrix[i - 1][j] + 1 ); } } } return matrix[b.length][a.length]; } function erstelleNeuenArtikel(artikelId, name, code, lagerplatzId, bestand) { const sheetArtikel = SpreadsheetApp.getActive().getSheetByName(SHEET_ARTIKEL); const sheetLager = SpreadsheetApp.getActive().getSheetByName(SHEET_LAGER); bestand = Number(bestand) || 0; if (lagerplatzId) { const lagerData = sheetLager.getDataRange().getValues(); let lagerExists = false; for (let i = 1; i < lagerData.length; i++) { if (String(lagerData[i][0]).trim().toUpperCase() === String(lagerplatzId).trim().toUpperCase()) { lagerExists = true; break; } } if (!lagerExists) { sheetLager.appendRow([lagerplatzId, lagerplatzId, 'Automatisch angelegt', 'JA']); } } sheetArtikel.appendRow([ artikelId, name, code || '', code || '', lagerplatzId || '', bestand, 1, 'JA', new Date() ]); if (bestand > 0) { schreibeBewegung(artikelId, name, 'EINLAGERUNG', bestand, 0, bestand, 'Admin-Handy', 'Ersterfassung'); } return { success: true }; } function updateArtikelData(artikelId, name, code, lagerplatzId, bestand) { if (!artikelId) throw new Error('Artikel-ID fehlt.'); const sheetArtikel = SpreadsheetApp.getActive().getSheetByName(SHEET_ARTIKEL); const sheetLager = SpreadsheetApp.getActive().getSheetByName(SHEET_LAGER); const data = sheetArtikel.getDataRange().getValues(); // Falls ein neuer Lagerplatz übergeben wurde, in Lagerplaetze-Sheet anlegen wenn nötig if (lagerplatzId) { const lagerData = sheetLager.getDataRange().getValues(); let lagerExists = false; for (let i = 1; i < lagerData.length; i++) { if (String(lagerData[i][0]).trim().toUpperCase() === String(lagerplatzId).trim().toUpperCase()) { lagerExists = true; break; } } if (!lagerExists) { sheetLager.appendRow([lagerplatzId, lagerplatzId, 'Automatisch angelegt', 'JA']); } } for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(artikelId)) { const rowIndex = i + 1; const bestandVorher = Number(data[i][5]) || 0; const neuerBestand = (bestand !== undefined && bestand !== null && bestand !== '') ? Number(bestand) : bestandVorher; // Spalten: 2: Name, 3: Barcode, 4: QR-Code, 5: Lagerplatz-ID, 6: Bestand, 9: Update-Datum sheetArtikel.getRange(rowIndex, 2).setValue(name || data[i][1]); sheetArtikel.getRange(rowIndex, 3).setValue(code !== undefined ? code : data[i][2]); sheetArtikel.getRange(rowIndex, 4).setValue(code !== undefined ? code : data[i][3]); sheetArtikel.getRange(rowIndex, 5).setValue(lagerplatzId !== undefined ? lagerplatzId : data[i][4]); sheetArtikel.getRange(rowIndex, 6).setValue(neuerBestand); sheetArtikel.getRange(rowIndex, 9).setValue(new Date()); // Wenn der Bestand manuell korrigiert wurde, Bewegung protokollieren if (neuerBestand !== bestandVorher) { const differenz = neuerBestand - bestandVorher; const typ = differenz > 0 ? 'EINLAGERUNG' : 'ENTNAHME'; schreibeBewegung( artikelId, name || data[i][1], typ, Math.abs(differenz), bestandVorher, neuerBestand, 'Admin', 'Manuelle Artikelkorrektur' ); } return { success: true }; } } throw new Error('Artikel nicht gefunden.'); } function entnahmeBuchen(artikelId, menge, benutzer, bemerkung) { menge = Number(menge); if (menge <= 0) throw new Error('Menge muss größer als 0 sein'); const lock = LockService.getScriptLock(); if (!lock.tryLock(5000)) { throw new Error('Parallele Verarbeitung. Bitte kurz warten.'); } try { const cache = CacheService.getScriptCache(); const lockKey = "LOCK_ENTNAHME_" + artikelId; if (cache.get(lockKey) !== null) { throw new Error('Doppelbuchung innerhalb kurzer Zeit abgefangen.'); } const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_ARTIKEL); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(artikelId)) { const bestandVorher = Number(data[i][5]); if (bestandVorher < menge) throw new Error('Nicht genügend Bestand'); const bestandNachher = bestandVorher - menge; sheet.getRange(i + 1, 6).setValue(bestandNachher); sheet.getRange(i + 1, 9).setValue(new Date()); schreibeBewegung(artikelId, data[i][1], 'ENTNAHME', menge, bestandVorher, bestandNachher, benutzer, bemerkung); cache.put(lockKey, "1", 3); return { success: true, bestand: bestandNachher }; } } throw new Error('Artikel nicht gefunden'); } finally { lock.releaseLock(); } } function einlagerungBuchen(artikelId, menge, benutzer, bemerkung) { menge = Number(menge); if (menge <= 0) throw new Error('Menge muss größer als 0 sein'); const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_ARTIKEL); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(artikelId)) { const bestandVorher = Number(data[i][5]); const bestandNachher = bestandVorher + menge; sheet.getRange(i + 1, 6).setValue(bestandNachher); sheet.getRange(i + 1, 9).setValue(new Date()); schreibeBewegung(artikelId, data[i][1], 'EINLAGERUNG', menge, bestandVorher, bestandNachher, benutzer, bemerkung); return { success: true, bestand: bestandNachher }; } } throw new Error('Artikel nicht gefunden'); } function schreibeBewegung(artikelId, artikelname, typ, menge, bestandVorher, bestandNachher, benutzer, bemerkung) { const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_BEWEGUNGEN); const id = sheet.getLastRow(); sheet.appendRow([ id, new Date(), artikelId, artikelname, typ, menge, bestandVorher, bestandNachher, benutzer, bemerkung ]); }
